// ══════════════════════════════════════

// СОСТОЯНИЕ ИГРЫ

let key_found = false;

let door_opened = false;

let gameStarted = false;

let audioCtx = null;


// ══════════════════════════════════════

// ИНИЦИАЛИЗАЦИЯ АУДИО (по жесту пользователя)

// ══════════════════════════════════════

function initAudio() {

  if (!audioCtx) {

    audioCtx = new (window.AudioContext || window.webkitAudioContext)();

  }

  if (audioCtx.state === 'suspended') {

    audioCtx.resume();

  }

}


// ══════════════════════════════════════

// ЗВУКОВЫЕ ЭФФЕКТЫ (Web Audio API)

// ══════════════════════════════════════

function playBell() {

  if (!audioCtx) return;

  const t = audioCtx.currentTime;

  // Колокольчик: два осциллятора

  [1200, 1800].forEach((freq, i) => {

    const osc = audioCtx.createOscillator();

    const gain = audioCtx.createGain();

    osc.connect(gain);

    gain.connect(audioCtx.destination);

    osc.frequency.value = freq;

    osc.type = 'sine';

    gain.gain.setValueAtTime(0.15, t + i * 0.1);

    gain.gain.exponentialRampToValueAtTime(0.001, t + 0.6 + i * 0.1);

    osc.start(t + i * 0.1);

    osc.stop(t + 0.7 + i * 0.1);

  });

}


function playKnock() {

  if (!audioCtx) return;

  const t = audioCtx.currentTime;

  // Тук-тук: короткие низкие удары

  [0, 0.2].forEach(delay => {

    const osc = audioCtx.createOscillator();

    const gain = audioCtx.createGain();

    osc.connect(gain);

    gain.connect(audioCtx.destination);

    osc.frequency.value = 150;

    osc.type = 'triangle';

    gain.gain.setValueAtTime(0.3, t + delay);

    gain.gain.exponentialRampToValueAtTime(0.001, t + delay + 0.15);

    osc.start(t + delay);

    osc.stop(t + delay + 0.2);

  });

}


function playDoorOpen() {

  if (!audioCtx) return;

  const t = audioCtx.currentTime;

  // Скрип двери + волшебный звук

  const osc = audioCtx.createOscillator();

  const gain = audioCtx.createGain();

  osc.connect(gain);

  gain.connect(audioCtx.destination);

  osc.frequency.setValueAtTime(200, t);

  osc.frequency.linearRampToValueAtTime(80, t + 1);

  osc.type = 'sawtooth';

  gain.gain.setValueAtTime(0.08, t);

  gain.gain.exponentialRampToValueAtTime(0.001, t + 1.2);

  osc.start(t);

  osc.stop(t + 1.3);


  // Волшебный аккорд

  [523, 659, 784].forEach((freq, i) => {

    const o = audioCtx.createOscillator();

    const g = audioCtx.createGain();

    o.connect(g);

    g.connect(audioCtx.destination);

    o.frequency.value = freq;

    o.type = 'sine';

    g.gain.setValueAtTime(0, t + 0.5 + i * 0.15);

    g.gain.linearRampToValueAtTime(0.12, t + 0.7 + i * 0.15);

    g.gain.exponentialRampToValueAtTime(0.001, t + 2.5);

    o.start(t + 0.5 + i * 0.15);

    o.stop(t + 2.6);

  });

}


function playFanfare() {

  if (!audioCtx) return;

  const t = audioCtx.currentTime;

  const notes = [523, 659, 784, 1047];

  notes.forEach((freq, i) => {

    const osc = audioCtx.createOscillator();

    const gain = audioCtx.createGain();

    osc.connect(gain);

    gain.connect(audioCtx.destination);

    osc.frequency.value = freq;

    osc.type = 'square';

    gain.gain.setValueAtTime(0.1, t + i * 0.2);

    gain.gain.exponentialRampToValueAtTime(0.001, t + i * 0.2 + 0.5);

    osc.start(t + i * 0.2);

    osc.stop(t + i * 0.2 + 0.6);

  });

}

let bubbleTimeout = null;


function showBubble(text, duration = 3500) {

  const bubble = document.getElementById('speechBubble');

  bubble.textContent = text;

  bubble.classList.add('visible');

  clearTimeout(bubbleTimeout);

  bubbleTimeout = setTimeout(() => {

    bubble.classList.remove('visible');

  }, duration);

}

function clickKey() {

  if (!gameStarted || key_found || door_opened) return;

  initAudio();


  key_found = true;


  // Визуал: ключ улетает

  const key = document.getElementById('keyObj');

  key.classList.add('found');

  document.getElementById('keySparkles').style.display = 'none';


  // Звук

  playBell();


  // Алиса прыгает

  const alice = document.getElementById('alice');

  alice.classList.add('alice-jump');

  setTimeout(() => alice.classList.remove('alice-jump'), 800);


  // Подсказка

  showBubble('🔑 Ключик найден! Но какую же дверь он открывает? 🚪', 4000);


  // Через секунду подсвечиваем золотую дверь

  setTimeout(() => {

    if (!door_opened) {

      const targetDoor = document.querySelector('.door-target');

      targetDoor.querySelector('.door-body').style.boxShadow = 

        '0 0 30px rgba(255,215,0,0.8), 0 0 60px rgba(255,215,0,0.4)';

    }

  }, 2500);

}

function clickDoor(doorEl) {

  if (!gameStarted || door_opened) return;

  initAudio();


  const type = doorEl.dataset.type;


  // ── ПРАВИЛО 2: Целевая дверь + ключ найден → УСПЕХ ──

  if (type === 'target' && key_found) {

    door_opened = true;


    // Дверь открывается

    doorEl.classList.add('door-opening');

    playDoorOpen();


    // Алиса идёт

    setTimeout(() => {

      const alice = document.getElementById('alice');

      alice.classList.add('walking');

      showBubble('🎉 Ура! Алиса идёт в Страну Чудес!', 3000);

    }, 800);


    // Финальный экран

    setTimeout(() => {

      playFanfare();

      document.getElementById('finalScreen').classList.add('visible');

    }, 3000);


    return;

  }


  // ── ПРАВИЛО 3: Целевая дверь, но ключ НЕ найден ──

  if (type === 'target' && !key_found) {

    playKnock();

    doorEl.classList.add('door-shake');

    setTimeout(() => doorEl.classList.remove('door-shake'), 600);

    showBubble('🔒 Дверь заперта на волшебный замок! Мне нужно найти ключик!', 3500);


    // Подсказка: подсветить ключ

    setTimeout(() => {

      const key = document.getElementById('keyObj');

      if (!key_found) key.classList.add('hint-glow');

      setTimeout(() => key.classList.remove('hint-glow'), 3000);

    }, 1500);

    return;

  }


  // ── ПРАВИЛО 4: Декоративные двери ──

  if (type === 'deco') {

    playKnock();

    doorEl.classList.add('door-shake');

    setTimeout(() => doorEl.classList.remove('door-shake'), 600);


    const messages = [

      '🚪 Эта дверь не открывается... Поищи другую!',

      '🤔 Хм, тут нет замочной скважины...',

      '🐇 «Не та дверь!» — шепчет Белый Кролик.',

      '🔒 Заперто! Ищи дверь с золотой скважиной!'

    ];

    showBubble(messages[Math.floor(Math.random() * messages.length)], 3000);

  }

}
